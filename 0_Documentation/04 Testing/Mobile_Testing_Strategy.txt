[TestFixture]
[Category("MobilePerformance")]
public class MobilePerformanceTests : BaseMobileTest
{
    [Test]
    public async Task PageLoad_3GNetwork_MeetsPerformanceTargets()
    {
        // Simulate 3G network conditions
        await SetNetworkConditions(NetworkSpeed.ThreeG);
        
        var performanceMetrics = new List<PagePerformanceMetric>();
        var testPages = new[] { "/", "/certificates", "/login", "/search" };
        
        foreach (var page in testPages)
        {
            var metrics = await MeasurePagePerformance(page);
            performanceMetrics.Add(metrics);
            
            // Assert performance targets
            metrics.PageLoadTime.Should().BeLessThan(TimeSpan.FromSeconds(3), 
                $"Page {page} should load within 3 seconds on 3G");
            
            metrics.FirstContentfulPaint.Should().BeLessThan(TimeSpan.FromSeconds(2),
                $"FCP for {page} should be under 2 seconds");
            
            metrics.LargestContentfulPaint.Should().BeLessThan(TimeSpan.FromSeconds(2.5),
                $"LCP for {page} should be under 2.5 seconds");
            
            metrics.TimeToInteractive.Should().BeLessThan(TimeSpan.FromSeconds(5),
                $"TTI for {page} should be under 5 seconds");
        }
        
        // Generate performance report
        await GeneratePerformanceReport(performanceMetrics);
    }
    
    [Test]
    public async Task ResourceLoading_MobileOptimization_EfficientlyLoaded()
    {
        await SetNetworkConditions(NetworkSpeed.ThreeG);
        await NavigateToPage("/certificates");
        
        // Measure resource loading
        var resourceMetrics = await GetResourceLoadingMetrics();
        
        // Verify image optimization
        var images = resourceMetrics.Where(r => r.ResourceType == "image");
        images.Should().AllSatisfy(img => 
        {
            img.FileSize.Should().BeLessThan(500 * 1024, "Images should be under 500KB");
            img.Format.Should().BeOneOf("webp", "jpeg", "png", "svg");
        });
        
        // Verify JavaScript bundling
        var scripts = resourceMetrics.Where(r => r.ResourceType == "script");
        scripts.Count().Should().BeLessThan(10, "Should have reasonable number of script files");
        
        // Verify CSS optimization
        var stylesheets = resourceMetrics.Where(r => r.ResourceType == "stylesheet");
        stylesheets.Should().AllSatisfy(css => 
            css.IsMinified.Should().BeTrue("CSS should be minified"));
        
        // Verify gzip compression
        resourceMetrics.Should().AllSatisfy(resource =>
            resource.IsCompressed.Should().BeTrue("Resources should be compressed"));
    }
    
    [Test]
    public async Task OfflineMode_ServiceWorker_CachesKeyResources()
    {
        // First, load page online to populate cache
        await NavigateToPage("/");
        await WaitForServiceWorkerRegistration();
        
        // Go offline
        await SetNetworkConditions(NetworkSpeed.Offline);
        
        // Attempt to reload page
        await RefreshPage();
        
        // Verify page loads from cache
        var pageTitle = await GetPageTitle();
        pageTitle.Should().NotBeNullOrEmpty("Page should load from cache when offline");
        
        // Verify critical functionality available offline
        var searchForm = await GetElement("#search-form");
        searchForm.Should().NotBeNull("Search form should be available offline");
        
        // Test cached certificate data access
        await TestOfflineCertificateAccess();
    }
    
    private async Task<PagePerformanceMetric> MeasurePagePerformance(string url)
    {
        var stopwatch = Stopwatch.StartNew();
        
        // Navigate to page
        await Driver.NavigateAsync(url);
        
        // Wait for page to load
        await WaitForPageLoadComplete();
        stopwatch.Stop();
        
        // Collect performance metrics using Performance API
        var performanceScript = @"
            return {
                pageLoadTime: performance.timing.loadEventEnd - performance.timing.navigationStart,
                domContentLoaded: performance.timing.domContentLoadedEventEnd - performance.timing.navigationStart,
                firstPaint: performance.getEntriesByName('first-paint')[0]?.startTime || 0,
                firstContentfulPaint: performance.getEntriesByName('first-contentful-paint')[0]?.startTime || 0,
                resourceCount: performance.getEntriesByType('resource').length
            };
        ";
        
        var metrics = await Driver.ExecuteScriptAsync<dynamic>(performanceScript);
        
        return new PagePerformanceMetric
        {
            Url = url,
            PageLoadTime = TimeSpan.FromMilliseconds(stopwatch.ElapsedMilliseconds),
            DomContentLoaded = TimeSpan.FromMilliseconds(metrics.domContentLoaded),
            FirstPaint = TimeSpan.FromMilliseconds(metrics.firstPaint),
            FirstContentfulPaint = TimeSpan.FromMilliseconds(metrics.firstContentfulPaint),
            ResourceCount = metrics.resourceCount
        };
    }
}

[TestFixture]
[Category("MobileSecurity")]
public class MobileSecurityTests : BaseMobileTest
{
    [Test]
    public async Task LocalStorage_SensitiveData_NotStoredInPlainText()
    {
        // Login to create session
        await LoginAsTestUser();
        
        // Check localStorage
        var localStorageScript = @"
            var storage = {};
            for (let i = 0; i < localStorage.length; i++) {
                let key = localStorage.key(i);
                storage[key] = localStorage.getItem(key);
            }
            return storage;
        ";
        
        var localStorage = await Driver.ExecuteScriptAsync<Dictionary<string, object>>(localStorageScript);
        
        // Check for common sensitive data patterns
        var sensitivePatterns = new[]
        {
            @"password", @"token", @"secret", @"key", @"auth",
            @"\d{4}-\d{4}-\d{4}-\d{4}", // Credit card pattern
            @"[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}", // Email pattern
            @"\d{3}-\d{2}-\d{4}" // SSN pattern
        };
        
        foreach (var item in localStorage)
        {
            var value = item.Value?.ToString() ?? "";
            
            foreach (var pattern in sensitivePatterns)
            {
                var regex = new Regex(pattern, RegexOptions.IgnoreCase);
                regex.IsMatch(value).Should().BeFalse(
                    $"LocalStorage item '{item.Key}' should not contain sensitive data pattern '{pattern}'");
            }
        }
        
        // Check sessionStorage similarly
        await CheckSessionStorageSecurity();
        
        // Check cookies
        await CheckCookieSecurity();
    }

    {
        await NavigateToPage("/certificates");
        
        // Get all interactive elements
        var interactiveElements = await GetElements(
            "button, a, input, select, [tabindex], [role='button'], [onclick]");
        
        var accessibilityIssues = new List<string>();
        
        foreach (var element in interactiveElements)

            var size = await element.GetSizeAsync();
            var location = await element.GetLocationAsync();
            
            // Check minimum size (44x44px)
            if (size.Height < 44 || size.Width < 44)
            {
                var elementInfo = await GetElementIdentifier(element);
                accessibilityIssues.Add($"Touch target too small: {elementInfo} ({size.Width}x{size.Height}px)");
            }
            
            // Check spacing between adjacent targets
            var adjacentElements = await GetAdjacentElements(element);
            foreach (var adjacent in adjacentElements)
            {
                var adjacentLocation = await adjacent.GetLocationAsync();
                var distance = CalculateDistance(location, adjacentLocation);
                
                if (distance < 8) // 8px minimum spacing
                {
                    var elementInfo = await GetElementIdentifier(element);
                    var adjacentInfo = await GetElementIdentifier(adjacent);
                    accessibilityIssues.Add($"Insufficient spacing between touch targets: {elementInfo} and {adjacentInfo} ({distance}px)");
                }
            }

    [Test]

        var headingLevels = new List<int>();
        foreach (var heading in headings)

            var tagName = await heading.GetTagNameAsync();
            var level = int.Parse(tagName.Substring(1));
            headingLevels.Add(level);
        }

        for (int i = 1; i < headingLevels.Count; i++)

        foreach (var element in ariaElements)
