```csharp
public class BaseApiClient
{
    protected readonly HttpClient _httpClient;
    protected readonly string _baseUrl;
    protected readonly ILogger _logger;
    
    public BaseApiClient(HttpClient httpClient, string baseUrl, ILogger logger)
    {
        _httpClient = httpClient;
        _baseUrl = baseUrl;
        _logger = logger;
    }
    
    protected async Task<ApiResponse<T>> ExecuteAsync<T>(ApiRequest request)
    {
        try
        {
            var httpRequest = BuildHttpRequestMessage(request);
            LogRequest(httpRequest);
            
            var stopwatch = Stopwatch.StartNew();
            var response = await _httpClient.SendAsync(httpRequest);
            stopwatch.Stop();
            
            var apiResponse = await ParseResponseAsync<T>(response);
            apiResponse.ResponseTime = stopwatch.Elapsed;
            
            LogResponse(apiResponse);
            return apiResponse;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "API request failed: {Method} {Url}", 
                request.Method, request.Endpoint);
            throw;
        }
    }
    
    private HttpRequestMessage BuildHttpRequestMessage(ApiRequest request)
    {
        var httpRequest = new HttpRequestMessage(request.Method, $"{_baseUrl}{request.Endpoint}");
        
        // Add headers
        foreach (var header in request.Headers)
        {
            httpRequest.Headers.Add(header.Key, header.Value);
        }
        
        // Add authentication
        if (!string.IsNullOrEmpty(request.AuthToken))
        {
            httpRequest.Headers.Authorization = 
                new AuthenticationHeaderValue("Bearer", request.AuthToken);
        }
        
        // Add content
        if (request.Content != null)
        {
            httpRequest.Content = new StringContent(
                JsonSerializer.Serialize(request.Content),
                Encoding.UTF8,
                "application/json");
        }
        
        return httpRequest;
    }
}
```

AUTHENTICATION HELPER:

```csharp
public class AuthenticationHelper
{
    private readonly BaseApiClient _authClient;
    private static readonly Dictionary<string, AuthToken> _tokenCache = new();
    
    public async Task<string> GetAuthTokenAsync(UserRole role = UserRole.Admin)
    {
        var cacheKey = role.ToString();
        
        if (_tokenCache.TryGetValue(cacheKey, out var cachedToken) && 
            cachedToken.ExpiresAt > DateTime.UtcNow.AddMinutes(5))
        {
            return cachedToken.AccessToken;
        }
        
        var credentials = GetCredentialsForRole(role);
        var loginRequest = new LoginRequest
        {
            Email = credentials.Email,
            Password = credentials.Password
        };
        
        var response = await _authClient.ExecuteAsync<LoginResponse>(new ApiRequest
        {
            Method = HttpMethod.Post,
            Endpoint = "/api/auth/login",
            Content = loginRequest
        });
        
        if (response.IsSuccess)
        {
            var authToken = new AuthToken
            {
                AccessToken = response.Data.AccessToken,
                ExpiresAt = DateTime.UtcNow.AddSeconds(response.Data.ExpiresIn)
            };
            
            _tokenCache[cacheKey] = authToken;
            return authToken.AccessToken;
        }
        
        throw new Exception($"Failed to authenticate as {role}: {response.Error}");
    }
    
    private static UserCredentials GetCredentialsForRole(UserRole role)
    {
        return role switch
        {
            UserRole.Admin => new("admin@gcc.com", "AdminPass123!"),
            UserRole.Analyst => new("analyst@gcc.com", "AnalystPass123!"),
            UserRole.Client => new("client@company.com", "ClientPass123!"),
            UserRole.Partner => new("partner@program.com", "PartnerPass123!"),
            UserRole.Public => new(null, null), // No auth needed
            _ => throw new ArgumentException($"Unknown role: {role}")
        };
    }
}
```
