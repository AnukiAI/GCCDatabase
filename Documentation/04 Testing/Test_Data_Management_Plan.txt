public class TestDataFactory
{
    private readonly Faker _faker;
    private readonly Random _random;
    
    public TestDataFactory(int seed = 12345)
    {
        _random = new Random(seed);
        _faker = new Faker();
        Randomizer.Seed = new Random(seed);
    }
    
    // Company Data Generation
    public Company GenerateCompany(int? id = null)
    {
        return new Company
        {
            Id = id ?? _random.Next(1, 10000),
            Name = _faker.Company.CompanyName(),
            ContactInfo = _faker.Internet.Email(),
            Address = _faker.Address.FullAddress(),
            Country = _faker.Address.CountryCode(),
            Phone = _faker.Phone.PhoneNumber(),
            Website = _faker.Internet.Url(),
            IsActive = _faker.Random.Bool(0.9f), // 90% active
            CreatedDate = _faker.Date.Between(DateTime.Now.AddYears(-5), DateTime.Now.AddMonths(-1))
        };
    }
    
    // Certificate Data Generation
    public Certificate GenerateCertificate(int? companyId = null, string status = null)
    {
        var effectiveDate = _faker.Date.Between(DateTime.Now.AddYears(-2), DateTime.Now.AddMonths(1));
        var expirationDate = effectiveDate.AddMonths(_faker.Random.Int(6, 36));
        
        return new Certificate
        {
            Id = _random.Next(1, 100000),
            Number = GenerateCertificateNumber(),
            CompanyId = companyId ?? _random.Next(1, 100),
            CertificationTypeId = _faker.Random.Int(1, 10),
            ProductId = _faker.Random.Int(1, 1000),
            Status = status ?? _faker.Random.WeightedRandom(
                new[] { "Active", "Expired", "Pending", "Suspended" },
                new[] { 0.6f, 0.2f, 0.15f, 0.05f }),
            EffectiveDate = effectiveDate,
            ExpirationDate = expirationDate,
            IssueDate = effectiveDate.AddDays(-_faker.Random.Int(1, 30)),
            Notes = _faker.Random.Bool(0.3f) ? _faker.Lorem.Sentence() : null,
            Asin = _faker.Random.Bool(0.4f) ? _faker.Random.AlphaNumeric(10).ToUpper() : null,
            Upc = _faker.Random.Bool(0.3f) ? _faker.Random.Digits(12) : null,
            CreatedDate = effectiveDate.AddDays(-_faker.Random.Int(1, 60))
        };
    }
    
    // User Data Generation
    public User GenerateUser(UserRole role = UserRole.Client, int? companyId = null)
    {
        var firstName = _faker.Name.FirstName();
        var lastName = _faker.Name.LastName();
        
        return new User
        {
            Id = _random.Next(1, 10000),
            Email = _faker.Internet.Email(firstName, lastName),
            FirstName = firstName,
            LastName = lastName,
            Role = role.ToString(),
            CompanyId = companyId ?? (role == UserRole.Client ? _random.Next(1, 100) : null),
            IsActive = _faker.Random.Bool(0.95f),
            EmailVerified = _faker.Random.Bool(0.9f),
            LastLogin = _faker.Date.Between(DateTime.Now.AddMonths(-3), DateTime.Now),
            CreatedDate = _faker.Date.Between(DateTime.Now.AddYears(-2), DateTime.Now.AddMonths(-1))
        };
    }
    
    // Product Data Generation
    public Product GenerateProduct(int? categoryId = null)
    {
        return new Product
        {
            Id = _random.Next(1, 10000),
            Name = _faker.Commerce.ProductName(),
            Description = _faker.Commerce.ProductDescription(),
            CategoryId = categoryId ?? _faker.Random.Int(1, 50),
            Sku = _faker.Commerce.Ean13(),
            Upc = _faker.Random.Bool(0.6f) ? _faker.Random.Digits(12) : null,
            Gtin = _faker.Random.Bool(0.4f) ? _faker.Random.Digits(14) : null,
            IsActive = _faker.Random.Bool(0.9f),
            CreatedDate = _faker.Date.Between(DateTime.Now.AddYears(-3), DateTime.Now)
        };
    }
    
    private string GenerateCertificateNumber()
    {
        var prefix = _faker.Random.WeightedRandom(
            new[] { "CERT", "ISO", "GMP", "HACCP", "FDA", "CE" },
            new[] { 0.3f, 0.2f, 0.2f, 0.1f, 0.1f, 0.1f });
        var year = DateTime.Now.Year;
        var sequence = _faker.Random.Int(1, 9999).ToString("D4");
        
        return $"{prefix}-{year}-{sequence}";
    }
    
    // Bulk Data Generation
    public List<T> GenerateList<T>(Func<T> generator, int count)
    {
        return Enumerable.Range(0, count).Select(_ => generator()).ToList();
    }
    
    // Scenario-Specific Data Sets
    public TestDataSet GenerateScenarioData(TestScenario scenario)
    {
        return scenario switch
        {
            TestScenario.NormalLoad => GenerateNormalLoadData(),
            TestScenario.HighVolume => GenerateHighVolumeData(),

            TestScenario.MultiTenant => GenerateMultiTenantData(),
            TestScenario.EdgeCases => GenerateEdgeCaseData(),
            _ => throw new ArgumentException($"Unknown scenario: {scenario}")
        };
    }
}

            () => _dataFactory.GenerateProduct(), dataVolume.ProductCount);

public class DataRefreshService
{
    public async Task<DataRefreshResult> RefreshEnvironmentData(
        TestEnvironment environment, RefreshOptions options = null)
    {
        var result = new DataRefreshResult();
        var stopwatch = Stopwatch.StartNew();
        
        try
        {
            // 1. Backup current data (if requested)
            if (options?.CreateBackup == true)
            {
                await CreateDataBackup(environment);
                result.BackupCreated = true;
            }
            
            // 2. Clean existing data
            await CleanEnvironmentData(environment);
            result.DataCleaned = true;
            
            // 3. Provision fresh data
            var provisioner = new TestDataProvisioner();
            await provisioner.ProvisionEnvironmentData(environment);
            result.DataProvisioned = true;
            
            // 4. Verify data integrity
            var verificationResult = await VerifyDataIntegrity(environment);
            result.DataVerified = verificationResult.IsValid;
            result.VerificationErrors = verificationResult.Errors;
            
            // 5. Update refresh metadata
            await UpdateRefreshMetadata(environment, DateTime.UtcNow);
            
            result.Success = true;
            result.Duration = stopwatch.Elapsed;
            
            return result;
        }
        catch (Exception ex)
        {
            result.Success = false;
            result.Error = ex.Message;
            result.Duration = stopwatch.Elapsed;
            
            // Attempt to restore from backup if available
            if (options?.CreateBackup == true)
            {
                await RestoreFromBackup(environment);
            }
            
            throw;
        }
    }
    
    public async Task SchedulePeriodicRefresh(TestEnvironment environment, TimeSpan interval)
    {
        // Implementation for scheduled data refresh
        var timer = new Timer(async _ => 
        {
            try
            {
                await RefreshEnvironmentData(environment);
                _logger.LogInformation("Scheduled data refresh completed for {Environment}", environment);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Scheduled data refresh failed for {Environment}", environment);
            }
        }, null, TimeSpan.Zero, interval);
    }
}

            LastName = _faker.Name.LastName(),
            Role = user.Role,

            IsActive = user.IsActive,
            EmailVerified = user.EmailVerified,
            LastLogin = user.LastLogin,
            CreatedDate = user.CreatedDate
        };
    }
    
    private string GenerateAnonymousCompanyName(string originalName)
    {
        // Preserve company type but anonymize name
        var companyTypes = new[] { "Inc", "LLC", "Corp", "Ltd", "Co" };
        var hasCompanyType = companyTypes.Any(type => originalName.EndsWith(type));
        
        var anonymousName = _faker.Company.CompanyName();
        
        if (hasCompanyType)
        {
            var originalType = companyTypes.First(type => originalName.EndsWith(type));
            anonymousName = $"{anonymousName.Split(' ')[0]} {originalType}";
        }
        
        return anonymousName;
    }
    
    private string GenerateAnonymousEmail()
    {
        return _faker.Internet.Email(domain: "testdata.local");
    }
}

public class GdprCompliantDataManager

    public async Task<DataExportResult> ExportPersonalData(int userId)

        using var context = new TestDbContext();

            .Where(a => a.UserId == userId)
            .ToListAsync();
        
        return new DataExportResult
        {
            PersonalData = new
            {
                User = user,
                Certificates = certificates,
                AuditLogs = auditLogs
            },
            ExportDate = DateTime.UtcNow,
            Format = "JSON"
        };
    }
    
    public async Task<DataDeletionResult> DeletePersonalData(int userId)
    {
        // Implement right to erasure
        using var context = new TestDbContext();
        using var transaction = await context.Database.BeginTransactionAsync();
        
        try
        {
            // Anonymize instead of delete to preserve referential integrity
            var user = await context.Users.FindAsync(userId);
            if (user != null)
            {
                user.Email = $"deleted-user-{userId}@anonymous.local";
                user.FirstName = "Deleted";
                user.LastName = "User";
                user.IsActive = false;
                user.EmailVerified = false;
            }
            
            // Update audit logs to remove personal identifiers
            var auditLogs = await context.AuditLogs
                .Where(a => a.UserId == userId)
                .ToListAsync();
                
            foreach (var log in auditLogs)
            {
                log.UserEmail = null;
                log.UserName = "Deleted User";
            }
            
            await context.SaveChangesAsync();
            await transaction.CommitAsync();
            
            return new DataDeletionResult { Success = true };
        }
        catch (Exception ex)
        {
            await transaction.RollbackAsync();
            return new DataDeletionResult { Success = false, Error = ex.Message };
        }
    }

public class TestDataIsolationManager
{
    private readonly ConcurrentDictionary<string, TestDataScope> _activeScopes = new();
    
    public async Task<TestDataScope> CreateIsolatedScope(string testName)
    {
        var scopeId = Guid.NewGuid().ToString();
        var scope = new TestDataScope
        {
            Id = scopeId,
            TestName = testName,
            CreatedAt = DateTime.UtcNow,
            IsolatedData = new Dictionary<Type, List<object>>()
        };
        
        _activeScopes[scopeId] = scope;
        
        // Create isolated test database or use table prefixing
        await CreateIsolatedDatabase(scopeId);
        
        return scope;
    }
    
    public async Task CleanupScope(TestDataScope scope)
    {
        try
        {
            // Clean up isolated data
            await CleanupIsolatedData(scope);
            
            // Remove from active scopes
            _activeScopes.TryRemove(scope.Id, out _);
            
            _logger.LogInformation("Test data scope {ScopeId} cleaned up successfully", scope.Id);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to cleanup test data scope {ScopeId}", scope.Id);
        }
    }
    
    private async Task CreateIsolatedDatabase(string scopeId)
    {
        var connectionString = GetTestConnectionString(scopeId);
        
        using var context = new TestDbContext(connectionString);
        await context.Database.EnsureCreatedAsync();
        
        // Apply any pending migrations
        await context.Database.MigrateAsync();
    }
    
    private async Task CleanupIsolatedData(TestDataScope scope)
    {
        var connectionString = GetTestConnectionString(scope.Id);
        
        using var context = new TestDbContext(connectionString);
        await context.Database.EnsureDeletedAsync();
    }
}

public class ParallelTestDataManager
{
    private readonly SemaphoreSlim _dataLock = new(1, 1);
    private readonly ConcurrentDictionary<int, TestDataPartition> _partitions = new();
    
    public async Task<TestDataPartition> AcquireDataPartition(string testCategory)
    {
        await _dataLock.WaitAsync();
        try
        {
            // Find available partition or create new one
            var partition = _partitions.Values.FirstOrDefault(p => 
                p.Category == testCategory && !p.IsInUse) 
                ?? await CreateNewPartition(testCategory);
            
            partition.IsInUse = true;
            partition.AssignedTest = Thread.CurrentThread.Name;
            
            return partition;
        }
        finally
        {
            _dataLock.Release();
        }
    }
    
    public async Task ReleaseDataPartition(TestDataPartition partition)
    {
        await _dataLock.WaitAsync();
        try
        {
            partition.IsInUse = false;
            partition.AssignedTest = null;
            
            // Clean up partition data if needed
            if (partition.RequiresCleanup)
            {
                await CleanupPartitionData(partition);
                partition.RequiresCleanup = false;
            }
        }
        finally
        {
            _dataLock.Release();
        }
    }
    
    private async Task<TestDataPartition> CreateNewPartition(string category)
    {
        var partitionId = _partitions.Count + 1;
        var partition = new TestDataPartition
        {
            Id = partitionId,
            Category = category,
            DatabaseName = $"TestDB_{category}_{partitionId}",
            DataRange = new DataRange
            {
                CompanyIdStart = partitionId * 1000,
                CompanyIdEnd = (partitionId + 1) * 1000 - 1,
                UserIdStart = partitionId * 10000,
                UserIdEnd = (partitionId + 1) * 10000 - 1
            }
        };
        
        await InitializePartitionData(partition);
        _partitions[partitionId] = partition;
        
        return partition;
    }
}

            Companies = await context.Companies.CountAsync(),
            Users = await context.Users.CountAsync(),
            Certificates = await context.Certificates.CountAsync(),
            Products = await context.Products.CountAsync()
        };
        
        ValidateCount("Companies", actualCounts.Companies, expectedVolumes.Companies, result);
        ValidateCount("Users", actualCounts.Users, expectedVolumes.Users, result);
        ValidateCount("Certificates", actualCounts.Certificates, expectedVolumes.Certificates, result);
        ValidateCount("Products", actualCounts.Products, expectedVolumes.Products, result);
    }
    
    private async Task ValidateDataIntegrity(DataValidationResult result)
    {
        using var context = new TestDbContext();
        
        // Check for orphaned records
        var orphanedCertificates = await context.Certificates
            .Where(c => !context.Companies.Any(comp => comp.Id == c.CompanyId))
            .CountAsync();
            
        if (orphanedCertificates > 0)
        {
            result.Errors.Add($"Found {orphanedCertificates} certificates with invalid company references");
        }
        
        var orphanedUsers = await context.Users
            .Where(u => u.CompanyId.HasValue && !context.Companies.Any(c => c.Id == u.CompanyId))
            .CountAsync();
            
        if (orphanedUsers > 0)
        {
            result.Errors.Add($"Found {orphanedUsers} users with invalid company references");
        }
        
        // Check for duplicate certificate numbers
        var duplicateCertNumbers = await context.Certificates
            .GroupBy(c => c.Number)
            .Where(g => g.Count() > 1)
            .CountAsync();
            
        if (duplicateCertNumbers > 0)
        {
            result.Errors.Add($"Found {duplicateCertNumbers} duplicate certificate numbers");
        }
    }
    
    private async Task ValidateBusinessRules(DataValidationResult result)
    {
        using var context = new TestDbContext();
        
        // Validate certificate date ranges
        var invalidDateRanges = await context.Certificates
            .Where(c => c.ExpirationDate <= c.EffectiveDate)
            .CountAsync();
            
        if (invalidDateRanges > 0)
        {
            result.Errors.Add($"Found {invalidDateRanges} certificates with invalid date ranges");
        }
        
        // Validate certificate validity periods (max 5 years)
        var excessiveValidityPeriods = await context.Certificates
            .Where(c => c.ExpirationDate > c.EffectiveDate.AddYears(5))
            .CountAsync();
            
        if (excessiveValidityPeriods > 0)
        {
            result.Errors.Add($"Found {excessiveValidityPeriods} certificates with validity periods exceeding 5 years");
        }
        
        // Validate email formats
        var invalidEmails = await context.Users
            .Where(u => !string.IsNullOrEmpty(u.Email) && !u.Email.Contains("@"))
            .CountAsync();
            
        if (invalidEmails > 0)
        {
            result.Errors.Add($"Found {invalidEmails} users with invalid email formats");
        }
    }
    
    private void ValidateCount(string entityName, int actual, int expected, DataValidationResult result)
    {
        var tolerance = expected * 0.1; // 10% tolerance
        if (Math.Abs(actual - expected) > tolerance)
        {
            result.Warnings.Add($"{entityName} count ({actual}) differs from expected ({expected}) by more than 10%");
        }
    }

public class TestDataBackupService

    public async Task<BackupResult> CreateBackup(TestEnvironment environment, string reason = null)
    {
        var backupId = Guid.NewGuid().ToString();
        var timestamp = DateTime.UtcNow;
        var backupPath = GetBackupPath(environment, timestamp);
        
        var result = new BackupResult
        {
            BackupId = backupId,
            Environment = environment,
            BackupPath = backupPath,
            CreatedAt = timestamp,
            Reason = reason
        };
        
        try
        {
            // Create database backup
            await CreateDatabaseBackup(environment, backupPath);
            result.DatabaseBackupSize = GetFileSize(Path.Combine(backupPath, "database.bak"));
            
            // Backup file attachments
            await BackupFileAttachments(environment, backupPath);
            result.FileBackupSize = GetDirectorySize(Path.Combine(backupPath, "files"));
            
            // Create metadata file
            await CreateBackupMetadata(result);
            
            result.Success = true;
            result.TotalSize = result.DatabaseBackupSize + result.FileBackupSize;
            
            // Register backup for cleanup scheduling
            await RegisterBackupForCleanup(result);
            
            _logger.LogInformation("Backup created successfully: {BackupId} for {Environment}", 
                backupId, environment);
        }
        catch (Exception ex)
        {
            result.Success = false;
            result.Error = ex.Message;
            
            // Cleanup partial backup
            await CleanupPartialBackup(backupPath);
            
            _logger.LogError(ex, "Backup failed for {Environment}: {BackupId}", environment, backupId);
        }
        
        return result;
    }
    
    public async Task<RestoreResult> RestoreFromBackup(string backupId)
    {
        var result = new RestoreResult { BackupId = backupId };
        
        try
        {
            var backup = await GetBackupMetadata(backupId);
            if (backup == null)
            {
                throw new InvalidOperationException($"Backup {backupId} not found");
            }
            
            // Restore database
            await RestoreDatabase(backup.Environment, backup.BackupPath);
            
            // Restore file attachments
            await RestoreFileAttachments(backup.Environment, backup.BackupPath);
            
            // Validate restored data
            var validation = await _validator.ValidateTestData(backup.Environment);
            if (!validation.IsValid)
            {
                throw new InvalidOperationException($"Restored data failed validation: {string.Join(", ", validation.Errors)}");
            }
            
            result.Success = true;
            result.RestoredAt = DateTime.UtcNow;
            
            _logger.LogInformation("Data restored successfully from backup {BackupId}", backupId);
        }
        catch (Exception ex)
        {
            result.Success = false;
            result.Error = ex.Message;
            
            _logger.LogError(ex, "Restore failed for backup {BackupId}", backupId);
        }
        
        return result;
    }
    
    public async Task ScheduleBackupCleanup(TimeSpan retentionPeriod)
    {
        var cutoffDate = DateTime.UtcNow - retentionPeriod;
        var oldBackups = await GetBackupsOlderThan(cutoffDate);
        
        foreach (var backup in oldBackups)
        {
            try
            {
                await DeleteBackup(backup.BackupId);
                _logger.LogInformation("Deleted old backup: {BackupId}", backup.BackupId);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to delete old backup: {BackupId}", backup.BackupId);
            }
        }
    }
}

public class PerformanceTestDataManager

            // Generate historical audit data
            await GenerateAuditHistory(config.HistoryMonths);
            
            // Create performance test indexes
            await CreatePerformanceIndexes();
            
            // Update statistics
            await UpdateDatabaseStatistics();
            
            stopwatch.Stop();
            
            _logger.LogInformation("High-volume data generation completed in {Duration}. " +
                "Generated {Companies} companies, {Users} users, {Products} products, {Certificates} certificates",
                stopwatch.Elapsed, config.CompanyCount, config.UserCount, 
                config.ProductCount, config.CertificateCount);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "High-volume data generation failed after {Duration}", stopwatch.Elapsed);
            throw;
        }
    }
    
    private async Task GenerateCompaniesInBatches(int totalCount, int batchSize)
    {
        var batches = (int)Math.Ceiling((double)totalCount / batchSize);
        
        for (int i = 0; i < batches; i++)
        {
            var currentBatchSize = Math.Min(batchSize, totalCount - (i * batchSize));
            var companies = _dataFactory.GenerateList(() => _dataFactory.GenerateCompany(), currentBatchSize);
            
            using var context = new TestDbContext();
            context.Companies.AddRange(companies);
            await context.SaveChangesAsync();
            
            if (i % 10 == 0) // Log progress every 10 batches
            {
                _logger.LogInformation("Generated {Count}/{Total} companies", (i + 1) * batchSize, totalCount);
            }
        }
    }
    
    private async Task CreatePerformanceIndexes()
    {
        using var context = new TestDbContext();
        var sql = @"
            -- Create covering index for certificate search
            CREATE INDEX IF NOT EXISTS IX_Certificates_Performance_Search 
            ON Certificates (CompanyId, Status, EffectiveDate, ExpirationDate) 
            INCLUDE (Number, CertificationTypeId, ProductId);
            
            -- Create index for user lookups
            CREATE INDEX IF NOT EXISTS IX_Users_Performance_Lookup 
            ON Users (CompanyId, Role, IsActive) 
            INCLUDE (Email, FirstName, LastName);
            
            -- Create index for audit queries
            CREATE INDEX IF NOT EXISTS IX_AuditLogs_Performance 
            ON AuditLogs (EntityId, EntityType, CreatedDate) 
            INCLUDE (Action, UserId);
        ";
        
        await context.Database.ExecuteSqlRawAsync(sql);
    }
