```csharp
public class ConfigurationValidator
{
    public ValidationResult ValidateConfiguration(IConfiguration config)
    {
        var result = new ValidationResult();
        
        // Validate database connection
        if (string.IsNullOrEmpty(config.GetConnectionString("DefaultConnection")))
        {
            result.Errors.Add("Database connection string is missing");
        }
        
        // Validate Azure AD B2C settings
        var b2cSection = config.GetSection("AzureAdB2C");
        if (string.IsNullOrEmpty(b2cSection["ClientId"]))
        {
            result.Errors.Add("Azure AD B2C ClientId is missing");
        }
        
        // Validate external service configurations
        ValidateExternalServices(config, result);
        
        return result;
    }
    
    private void ValidateExternalServices(IConfiguration config, ValidationResult result)
    {
        var emailConfig = config.GetSection("ExternalServices:EmailService");
        if (string.IsNullOrEmpty(emailConfig["ApiKey"]))
        {
            result.Errors.Add("Email service API key is missing");
        }
    }
}
```

```csharp
public class EnvironmentHealthMonitor
{
    private readonly ILogger<EnvironmentHealthMonitor> _logger;
    private readonly IConfiguration _configuration;
    
    public async Task<HealthCheckResult> CheckEnvironmentHealth()
    {
        var healthChecks = new List<Task<HealthCheckResult>>
        {
            CheckDatabaseConnectivity(),
            CheckExternalServices(),
            CheckApplicationHealth(),
            CheckStorageAvailability(),
            CheckAuthenticationService()
        };
        
        var results = await Task.WhenAll(healthChecks);
        
        return AggregateResults(results);
    }
    
    private async Task<HealthCheckResult> CheckDatabaseConnectivity()
    {
        try
        {
            using var connection = new SqlConnection(_configuration.GetConnectionString("DefaultConnection"));
            await connection.OpenAsync();
            return HealthCheckResult.Healthy("Database connection successful");
        }
        catch (Exception ex)
        {
            return HealthCheckResult.Unhealthy("Database connection failed", ex);
        }
    }
}
```

```csharp
public class AutomatedHealthChecker
{
    private readonly IServiceProvider _serviceProvider;
    
    public async Task RunScheduledHealthChecks()
    {
        var environments = GetAllEnvironments();
        
        foreach (var env in environments)
        {
            var healthResult = await CheckEnvironmentHealth(env);
            await ProcessHealthCheckResult(env, healthResult);
        }
    }
    
    private async Task ProcessHealthCheckResult(Environment env, HealthCheckResult result)
    {
        if (result.Status == HealthStatus.Unhealthy)
        {
            await SendAlert(env, result);
            await InitiateRecoveryProcedures(env, result);
        }
        
        await LogHealthCheckResult(env, result);
        await UpdateEnvironmentStatus(env, result);
    }
}
```
