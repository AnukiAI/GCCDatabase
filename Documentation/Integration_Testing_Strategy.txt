```csharp
public abstract class IntegrationTestBase
{
    protected TestApplication App { get; private set; }
    protected HttpClient Client { get; private set; }
    protected DatabaseContext Context { get; private set; }
    
    [OneTimeSetUp]
    public void OneTimeSetUp()
    {
        App = new TestApplicationFactory()
            .WithAuthentication()
            .WithDatabase()
            .WithExternalServices()
            .Create();
            
        Client = App.CreateClient();
        Context = App.GetService<DatabaseContext>();
    }
    
    [SetUp]
    public void SetUp()
    {
        Context.Database.BeginTransaction();
    }
    
    [TearDown]
    public void TearDown()
    {
        Context.Database.RollbackTransaction();
    }
}
```

```csharp
public class TestDataFactory
{
    private readonly Faker<Certificate> _certificateFaker;
    
    public TestDataFactory()
    {
        _certificateFaker = new Faker<Certificate>()
            .RuleFor(c => c.Number, f => f.Random.AlphaNumeric(10))
            .RuleFor(c => c.Status, f => f.PickRandom("Active", "Expired", "Pending"))
            .RuleFor(c => c.EffectiveDate, f => f.Date.Recent(30))
            .RuleFor(c => c.ExpirationDate, f => f.Date.Future(1));
    }
    
    public Certificate CreateCertificate() => _certificateFaker.Generate();
    public List<Certificate> CreateCertificates(int count) => _certificateFaker.Generate(count);
}
```
