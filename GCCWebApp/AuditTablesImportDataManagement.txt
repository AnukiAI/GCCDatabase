  // Soft delete
  public async Task<bool> SoftDeleteCertificateAsync(int id, int userId, string? reason)
  {
      var certificate = await _context.Certificates
          .IgnoreQueryFilters() // Include already deleted
          .FirstOrDefaultAsync(c => c.Id == id);

      if (certificate == null) return false;

      certificate.IsDeleted = true;
      certificate.DeletedAt = DateTime.UtcNow;
      certificate.DeletedBy = userId;
      certificate.DeletionReason = reason;
      certificate.UpdatedAt = DateTime.UtcNow;

      await _auditService.LogAsync("Certificate", id, "SoftDelete", userId, reason);
      await _context.SaveChangesAsync();
      return true;
  }

  // Restore deleted record
  public async Task<bool> RestoreCertificateAsync(int id, int userId)
  {
      var certificate = await _context.Certificates
          .IgnoreQueryFilters()
          .FirstOrDefaultAsync(c => c.Id == id && c.IsDeleted);

      if (certificate == null) return false;

      certificate.IsDeleted = false;
      certificate.DeletedAt = null;
      certificate.DeletedBy = null;
      certificate.DeletionReason = null;

      await _auditService.LogAsync("Certificate", id, "Restore", userId);
      await _context.SaveChangesAsync();
      return true;
  }

  3. Change History Tracking for All CRUD Operations

  Implementation Strategy

  Option A: Automatic via EF Core Interceptors (Recommended)

  Create a SaveChangesInterceptor that automatically logs all changes:

  public class AuditInterceptor : SaveChangesInterceptor
  {
      private readonly IHttpContextAccessor _httpContextAccessor;

      public override async ValueTask<InterceptionResult<int>> SavingChangesAsync(
          DbContextEventData eventData,
          InterceptionResult<int> result,
          CancellationToken cancellationToken = default)
      {
          var context = eventData.Context;
          if (context == null) return result;

          var entries = context.ChangeTracker.Entries()
              .Where(e => e.State == EntityState.Added ||
                         e.State == EntityState.Modified ||
                         e.State == EntityState.Deleted)
              .ToList();

          var userId = GetCurrentUserId();
          var ipAddress = GetClientIPAddress();

          foreach (var entry in entries)
          {
              var auditLog = new AuditLog
              {
                  EntityName = entry.Entity.GetType().Name,
                  EntityId = GetPrimaryKeyValue(entry),
                  Action = entry.State.ToString(),
                  UserId = userId,
                  Timestamp = DateTime.UtcNow,
                  IPAddress = ipAddress,
                  OldValues = GetOldValues(entry),
                  NewValues = GetNewValues(entry),
                  ChangedFields = GetChangedFields(entry)
              };

              context.Set<AuditLog>().Add(auditLog);
          }

          return result;
      }

      private string GetOldValues(EntityEntry entry)
      {
          if (entry.State == EntityState.Added) return null;

          var oldValues = new Dictionary<string, object>();
          foreach (var prop in entry.OriginalValues.Properties)
          {
              oldValues[prop.Name] = entry.OriginalValues[prop];
          }
          return JsonSerializer.Serialize(oldValues);
      }

      private string GetNewValues(EntityEntry entry)
      {
          if (entry.State == EntityState.Deleted) return null;

          var newValues = new Dictionary<string, object>();
          foreach (var prop in entry.CurrentValues.Properties)
          {
              newValues[prop.Name] = entry.CurrentValues[prop];
          }
          return JsonSerializer.Serialize(newValues);
      }
  }

  Register in Program.cs:
  builder.Services.AddDbContext<ApplicationDbContext>(options =>
  {
      options.UseSqlServer(connectionString)
             .AddInterceptors(new AuditInterceptor(httpContextAccessor));
  });

  Option B: Manual Audit Service

  public class AuditService
  {
      private readonly ApplicationDbContext _context;
      private readonly IHttpContextAccessor _httpContextAccessor;

      public async Task LogCreateAsync<T>(T entity, int userId) where T : class
      {
          var auditLog = new AuditLog
          {
              EntityName = typeof(T).Name,
              EntityId = GetEntityId(entity),
              Action = "Create",
              UserId = userId,
              Timestamp = DateTime.UtcNow,
              NewValues = JsonSerializer.Serialize(entity),
              IPAddress = GetClientIPAddress()
          };

          _context.AuditLogs.Add(auditLog);
          await _context.SaveChangesAsync();
      }

      public async Task LogUpdateAsync<T>(T oldEntity, T newEntity, int userId)
      {
          var changes = GetChanges(oldEntity, newEntity);

          var auditLog = new AuditLog
          {
              EntityName = typeof(T).Name,
              EntityId = GetEntityId(newEntity),
              Action = "Update",
              UserId = userId,
              Timestamp = DateTime.UtcNow,
              OldValues = JsonSerializer.Serialize(oldEntity),
              NewValues = JsonSerializer.Serialize(newEntity),
              ChangedFields = JsonSerializer.Serialize(changes.Keys.ToList()),
              IPAddress = GetClientIPAddress()
          };

          _context.AuditLogs.Add(auditLog);
          await _context.SaveChangesAsync();
      }

      public async Task LogDeleteAsync<T>(T entity, int userId, string? reason = null)
      {
          var auditLog = new AuditLog
          {
              EntityName = typeof(T).Name,
              EntityId = GetEntityId(entity),
              Action = "Delete",
              UserId = userId,
              Timestamp = DateTime.UtcNow,
              OldValues = JsonSerializer.Serialize(entity),
              Reason = reason,
              IPAddress = GetClientIPAddress()
          };

          _context.AuditLogs.Add(auditLog);
          await _context.SaveChangesAsync();
      }
  }
